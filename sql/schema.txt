Table users {
  id                integer [primary key]
  username          varchar [not null]
  email             varchar [not null]
  first_name        varchar [not null]
  last_name         varchar [not null]
  created_at        timestamp [not null, default: `now()`]
  updated_at        timestamp [not null, default: `now()`]
}

Table accounts {
  id                    integer [primary key]
  name                  varchar [not null]
  user_id               integer [not null]
  account_type          varchar [not null, note: 'checking | savings | credit_card | cash | other']
  starting_balance      numeric(12, 2) [not null, default: 0]
  is_archived           boolean [not null, default: false]
  created_at            timestamp [not null, default: `now()`]
  updated_at            timestamp [not null, default: `now()`]
}

Ref: accounts.user_id > users.id [delete: cascade]

Table categories {
  id           integer [primary key]
  user_id      integer
  name         varchar [not null]
  is_default   boolean [not null, default: false]
  created_at   timestamp [not null, default: `now()`]
}

Ref: categories.user_id > users.id [delete: cascade]

Table transactions {
  id                    integer [primary key]
  account_id            integer [not null]
  category_id           integer [note: 'nullable — uncategorized is valid']
  description           varchar [not null]
  amount                numeric(12,2) [not null, note: 'always positive; sign comes from transaction_type']
  transaction_type      varchar [not null, note: 'income | expense']
  transaction_date      date [not null]
  created_at            timestamp [not null, default: `now()`]
  updated_at            timestamp [not null, default: `now()`]
}

Ref: transactions.account_id > accounts.id [delete: cascade]
Ref: transactions.category_id > categories.id [delete: set null]

Table budgets {
  id                    integer [primary key]
  user_id               integer [not null]
  month                 integer [not null, note: '1–12']
  year                  integer [not null]
  created_at            timestamp [not null, default: `now()`]

  indexes {
    (user_id, month, year) [unique]
  }
}

Ref: budgets.user_id > users.id [delete: cascade]

Table budget_categories {
  id              integer [primary key]
  budget_id       integer [not null]
  category_id     integer [not null]
  limit_amount    numeric(12,2) [not null]

  indexes {
    (budget_id, category_id) [unique]
  }
}

Ref: budget_categories.budget_id > budgets.id [delete: cascade]
Ref: budget_categories.category_id > categories.id [delete: cascade]

Table savings_goals {
  id                integer [primary key]
  user_id           integer [not null]
  name              varchar [not null]
  target_amount     numeric(12,2) [not null]
  current_amount    numeric(12,2) [not null, default: 0]
  target_date       date [not null]
  is_completed      boolean [not null, default: false]
  created_at        timestamp [not null, default: `now()`]
  updated_at        timestamp [not null, default: `now()`]
}

Ref: savings_goals.user_id > users.id [delete: cascade]

Table goal_contributions {
  id                    integer [primary key]
  goal_id               integer [not null]
  amount                numeric(12,2) [not null]
  contribution_date     date [not null]
  created_at            timestamp [not null, default: `now()`]
}

Ref: goal_contributions.goal_id > savings_goals.id [delete: cascade]
